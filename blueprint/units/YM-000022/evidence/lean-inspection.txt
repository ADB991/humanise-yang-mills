def YM.Gauge.Simple : YM.Gauge → Prop :=
fun G =>
  (∃ X ∈ G.lie, ∃ Y ∈ G.lie, X * Y - Y * X ≠ 0) ∧
    ∀ I ≤ G.lie, (∀ X ∈ G.lie, ∀ Y ∈ I, X * Y - Y * X ∈ I) → I = ⊥ ∨ I = G.lie
def YM.Gauge.casimir : YM.Gauge → ℝ :=
fun G =>
  have b := EuclideanSpace.basisFun G.ColorIndex ℝ;
  (-∑ i, (LinearMap.trace ℝ G.Color) (G.ad (b i) ∘ₗ G.ad (b i))) / ↑G.lieDim
def YM.Gauge.slopeConstant : YM.Gauge → ℝ :=
fun G => 11 * G.casimir / (24 * Real.pi ^ 2)
